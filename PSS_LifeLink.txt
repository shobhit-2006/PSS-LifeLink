string toLower(string text)
{
    transform(text.begin(), text.end(), text.begin(),
        [](unsigned char ch)
        {
            return tolower(ch);
        });

    return text;
}


// BASE CLASS: Inheritance + Polymorphism
class HospitalResource
{
public:
    virtual void display() const = 0;

    virtual ~HospitalResource() = default;
};

// DERIVED CLASS: Organ
class Organ : public HospitalResource
{
private:
    string name, bloodGroup, status;
    int quantity;

public:
    Organ(string n, string b, int q, string s)
        : name(n), bloodGroup(b), quantity(q), status(s) {
    }

    void display() const override
    {
        cout << "\nResource Type: Organ\n";
        cout << "Organ: " << name << endl;
        cout << "Blood Group: " << bloodGroup << endl;
        cout << "Quantity: " << quantity << endl;
        cout << "Status: " << status << endl;
    }
};

// DERIVED CLASS: BloodBank
class BloodBank : public HospitalResource
{
private:
    string bloodGroup;
    int units;

public:
    BloodBank(string b, int u)
        : bloodGroup(b), units(u) {
    }

    void display() const override
    {
        cout << "\nResource Type: Blood Bank\n";
        cout << "Blood Group: " << bloodGroup << endl;
        cout << "Units Available: " << units << endl;
    }
};

// DATABASE CLASS
class DatabaseManager
{
private:
    unique_ptr<sql::Connection> con;

    template <typename Callback>
    void runProcedure(
        const string& procedure,
        const string& value,
        Callback callback)
    {
        unique_ptr<sql::PreparedStatement> stmt(
            con->prepareStatement(procedure)
        );

        stmt->setString(1, value);

        unique_ptr<sql::ResultSet> rs(
            stmt->executeQuery()
        );

        callback(rs.get());

        rs.reset();

        while (stmt->getMoreResults())
        {
            unique_ptr<sql::ResultSet> extra(
                stmt->getResultSet()
            );
        }
    }


public:
    DatabaseManager()
    {
        sql::mysql::MySQL_Driver* driver =
            sql::mysql::get_mysql_driver_instance();
        con.reset(
            driver->connect(
                "tcp://127.0.0.1:3306",
                "root",
                "Shobhit#26"
            )
        );

        con->setSchema("PSS_LifeLink");
    }


    // FUNCTION OVERLOADING
    // Search by Hospital ID
    void search(int hospitalID)
    {
        sql::PreparedStatement* stmt = nullptr;
        sql::ResultSet* rs = nullptr;

        try
        {
            stmt = con->prepareStatement(
                "CALL SearchHospitalByID(?)"
            );

            stmt->setInt(1, hospitalID);
            rs = stmt->executeQuery();

            bool found = false;

            while (rs->next())
            {
                if (!found)
                {
                    cout << "\n========== HOSPITAL DETAILS ==========\n";
                    cout << "Hospital ID: "
                        << rs->getInt("hospital_ID") << endl;

                    cout << "Hospital Name: "
                        << rs->getString("hospital_name") << endl;

                    cout << "City: "
                        << rs->getString("city") << endl;

                    cout << "Address: "
                        << rs->getString("address") << endl;

                    cout << "Phone: "
                        << rs->getString("hospital_phone") << endl;

                    cout << "Email: "
                        << rs->getString("hospital_email") << endl;

                    cout << "Authorization Status: "
                        << rs->getString("authorization_status") << endl;

                    cout << "\n========== ORGAN & BLOOD DETAILS ==========\n";

                    found = true;
                }

                string itemType = rs->getString("item_type");

                if (itemType == "Organ")
                {
                    cout << "\nType: Organ" << endl;

                    cout << "Organ Name: "
                        << rs->getString("item_name") << endl;

                    cout << "Blood Group: "
                        << rs->getString("blood_group") << endl;

                    cout << "Quantity: "
                        << rs->getInt("available_amount") << endl;

                    cout << "Availability: "
                        << rs->getString("current_status") << endl;
                }
                else if (itemType == "Blood")
                {
                    cout << "\nType: Blood Bank" << endl;

                    cout << "Blood Group: "
                        << rs->getString("item_name") << endl;

                    cout << "Units Available: "
                        << rs->getInt("available_amount") << endl;

                    cout << "Status: "
                        << rs->getString("current_status") << endl;
                }

                cout << "----------------------------------------\n";
            }

            if (!found)
            {
                cout << "Hospital ID not found or no linked records found.\n";
            }

            delete rs;
            rs = nullptr;

            delete stmt;
            stmt = nullptr;
        }
        catch (sql::SQLException& e)
        {
            delete rs;
            delete stmt;

            cout << "Database Error: " << e.what() << endl;
        }
    }



    // FUNCTION OVERLOADING
    // Search hospitals by city
    void search(const string& city)
    {
        sql::PreparedStatement* stmt = nullptr;
        sql::ResultSet* rs = nullptr;

        try
        {
            stmt = con->prepareStatement(
                "CALL SearchByCity(?)"
            );

            rs = stmt->executeQuery();

            bool found = false;
            int previousHospitalID = -1;

            while (rs->next())
            {
                int currentHospitalID =
                    rs->getInt("hospital_ID");

                // Print hospital details only once
                if (currentHospitalID != previousHospitalID)
                {
                    if (found)
                    {
                        cout << "\n========================================\n";
                    }

                    cout << "\n========== HOSPITAL DETAILS ==========\n";

                    cout << "Hospital ID: "
                        << currentHospitalID << endl;

                    cout << "Hospital Name: "
                        << rs->getString("hospital_name") << endl;

                    cout << "City: "
                        << rs->getString("city") << endl;

                    cout << "Address: "
                        << rs->getString("address") << endl;

                    cout << "Phone: "
                        << rs->getString("hospital_phone") << endl;

                    cout << "Email: "
                        << rs->getString("hospital_email") << endl;

                    cout << "Authorization Status: "
                        << rs->getString("authorization_status") << endl;

                    cout << "\n========== BLOOD BANK DETAILS ==========\n";

                    previousHospitalID = currentHospitalID;
                    found = true;
                }

                string itemType = rs->getString("item_type");

                if (itemType == "Blood")
                {
                    cout << "\nBlood Group: "
                        << rs->getString("item_name") << endl;

                    cout << "Units Available: "
                        << rs->getInt("available_amount") << endl;

                    cout << "Status: "
                        << rs->getString("current_status") << endl;
                }
                else if (itemType == "Organ")
                {
                    cout << "\n========== ORGAN DETAILS ==========\n";

                    cout << "Organ Name: "
                        << rs->getString("item_name") << endl;

                    cout << "Blood Group: "
                        << rs->getString("blood_group") << endl;

                    cout << "Quantity: "
                        << rs->getInt("available_amount") << endl;

                    cout << "Availability: "
                        << rs->getString("current_status") << endl;
                }

                cout << "----------------------------------------\n";
            }

            if (!found)
            {
                cout << "No records found for this city.\n";
            }

            delete rs;
            rs = nullptr;

            delete stmt;
            stmt = nullptr;
        }
        catch (sql::SQLException& e)
        {
            delete rs;
            delete stmt;

            cout << "Database Error: "
                << e.what() << endl;
        }
    }


    void searchOrgan(const string& organName)
    {
        runProcedure(
            "CALL SearchOrgan(?)", organName,
            [](sql::ResultSet* rs)
            {
                bool found = false;

                while (rs->next())
                {
                    found = true;

                    // Create Organ object using runtime polymorphism
                    unique_ptr<HospitalResource> resource =
                        make_unique<Organ>(
                            rs->getString("organ_name"),
                            rs->getString("blood_group"),
                            rs->getInt("quantity"),
                            rs->getString("availability_status")
                        );

                    // Display hospital details
                    cout << "\n========== HOSPITAL DETAILS ==========\n";

                    cout << "Hospital ID: "
                        << rs->getInt("hospital_ID") << endl;

                    cout << "Hospital Name: "
                        << rs->getString("hospital_name") << endl;

                    cout << "City: "
                        << rs->getString("city") << endl;

                    cout << "Address: "
                        << rs->getString("address") << endl;

                    cout << "Phone: "
                        << rs->getString("phone") << endl;

                    cout << "Email: "
                        << rs->getString("email") << endl;

                    // Display organ details
                    cout << "\n========== ORGAN DETAILS ==========\n";
                    resource->display();

                    cout << "====================================\n";
                }

                if (!found)
                {
                    cout << "No available organs found.\n";
                }
            }
        );
    }


    void searchBloodGroup(const string& group)
    {
        runProcedure(
            "CALL SearchBloodGroup(?)",
            group,
            [](sql::ResultSet* rs)
            {
                bool found = false;

                while (rs->next())
                {
                    found = true;

                    unique_ptr<HospitalResource> resource =
                        make_unique<BloodBank>(
                            rs->getString("blood_group"),
                            rs->getInt("units_available")
                        );

                    cout << "\n========== HOSPITAL DETAILS ==========\n";
                    cout << "Hospital ID: "
                        << rs->getInt("hospital_ID") << endl;
                    cout << "Hospital Name: "
                        << rs->getString("hospital_name") << endl;
                    cout << "City: "
                        << rs->getString("city") << endl;
                    cout << "Address: "
                        << rs->getString("address") << endl;
                    cout << "Phone: "
                        << rs->getString("phone") << endl;
                    cout << "Email: "
                        << rs->getString("email") << endl;

                    cout << "\n========== BLOOD BANK DETAILS ==========\n";
                    resource->display();
                    cout << "========================================\n";
                }

                if (!found)
                {
                    cout << "No blood stock found for this group.\n";
                }
            }
        );
    }

};
